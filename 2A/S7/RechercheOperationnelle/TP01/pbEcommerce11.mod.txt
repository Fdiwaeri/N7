###############################  Model ###############################


###############################  Sets  ###############################

set CLIENTS;
set MAGASINS;
set FLUIDES;

################### Variables ###################

var X{i in CLIENTS, j in MAGASINS, k in FLUIDES}, >=0; 
 

###################  Constants: Data to load   #########################

param commandes{i in CLIENTS, j in FLUIDES};
param stocks{i in MAGASINS, j in FLUIDES};
param couts_unit{i in MAGASINS, j in FLUIDES};

################### Constraints ###################


s.t. LimiteStock{j in MAGASINS, k in FLUIDES}:
  sum{i in CLIENTS} X[i,j,k] <= stocks[j,k];

s.t. commandeClient{i in CLIENTS , k in FLUIDES}:
  sum{j in MAGASINS} X[i,j,k] >= commandes[i,k];

###### Objective ######

minimize cout: 
		sum{i in CLIENTS, j in MAGASINS , k in FLUIDES} X[i,j,k]* couts_unit[j,k]; 


end;
